cmake_minimum_required(VERSION 3.22)
project(SelectedStructuralFaceKernel LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
add_library(selected_structural_face_kernel SHARED src/face_kernel.cpp src/structure.cpp src/register_types.cpp)
set_source_files_properties(src/face_kernel.cpp src/structure.cpp src/register_types.cpp PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/face_kernel.hpp")
target_include_directories(selected_structural_face_kernel PRIVATE "../build/godot-cpp/gen/include" "../../native-character-nativephysics/vendor/godot-cpp/include")
target_compile_definitions(selected_structural_face_kernel PRIVATE GDEXTENSION NOMINMAX THREADS_ENABLED TYPED_METHOD_BIND WINDOWS_ENABLED _HAS_EXCEPTIONS=0)
target_compile_features(selected_structural_face_kernel PRIVATE cxx_std_17)
target_compile_options(selected_structural_face_kernel PRIVATE /O2 /fp:precise /W4 /utf-8 /Zc:__cplusplus)
target_link_libraries(selected_structural_face_kernel PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../build/bin/libgodot-cpp.windows.template_release.x86_64.lib")
set_target_properties(selected_structural_face_kernel PROPERTIES RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_CURRENT_SOURCE_DIR}/godot/bin>" OUTPUT_NAME "selected_structural_face_kernel.windows.release.x86_64")

cmake_minimum_required(VERSION 3.22)
project(SelectedPaintedFaceKernel LANGUAGES CXX)
set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
set(GODOTCPP_API_VERSION "4.7" CACHE STRING "" FORCE)
set(GODOTCPP_CUSTOM_API_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../native-character-nativephysics/extension_api.json" CACHE PATH "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE PATH "" FORCE)
set(GODOTCPP_PRECISION single CACHE STRING "" FORCE)
add_subdirectory(../native-character-nativephysics/vendor/godot-cpp godot-cpp)
add_library(selected_face_kernel SHARED src/face_kernel.cpp src/register_types.cpp)
set_source_files_properties(src/face_kernel.cpp src/register_types.cpp PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/face_kernel.hpp")
target_link_libraries(selected_face_kernel PRIVATE godot::cpp)
target_compile_features(selected_face_kernel PRIVATE cxx_std_17)
if(MSVC)
 target_compile_options(selected_face_kernel PRIVATE /O2 /fp:precise /W4)
endif()
set_target_properties(selected_face_kernel PROPERTIES RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_CURRENT_SOURCE_DIR}/godot/bin>" LIBRARY_OUTPUT_DIRECTORY "$<1:${CMAKE_CURRENT_SOURCE_DIR}/godot/bin>" OUTPUT_NAME "selected_face_kernel.windows.release.x86_64")

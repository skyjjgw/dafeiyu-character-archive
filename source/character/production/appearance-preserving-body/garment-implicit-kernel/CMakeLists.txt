cmake_minimum_required(VERSION 3.22)
project(SelectedImplicitCloth LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
add_library(selected_implicit_cloth SHARED src/implicit_cloth.cpp src/register_types.cpp)
target_include_directories(selected_implicit_cloth PRIVATE "../../native-character-nativephysics/build/vendor/godot-cpp/gen/include" "../../native-character-nativephysics/vendor/godot-cpp/include")
target_compile_definitions(selected_implicit_cloth PRIVATE GDEXTENSION NOMINMAX THREADS_ENABLED TYPED_METHOD_BIND WINDOWS_ENABLED _HAS_EXCEPTIONS=0)
target_compile_features(selected_implicit_cloth PRIVATE cxx_std_17)
target_compile_options(selected_implicit_cloth PRIVATE /O2 /fp:precise /W4 /utf-8 /Zc:__cplusplus)
target_link_libraries(selected_implicit_cloth PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../native-character-nativephysics/build/bin/libgodot-cpp.windows.template_release.x86_64.lib")
set_target_properties(selected_implicit_cloth PROPERTIES RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_CURRENT_SOURCE_DIR}/project/bin>" OUTPUT_NAME "selected_implicit_cloth.windows.release.x86_64")
